dd_has_three_political_parties = {
	count_triggers = {
		amount > 2
		check_variable = { party_popularity@libertarian_socialism > 0.001 }
		check_variable = { party_popularity@syndicate_socialism > 0.001 }
		check_variable = { party_popularity@communism > 0.001 }
		check_variable = { party_popularity@social_democracy > 0.001 }
		check_variable = { party_popularity@democratic > 0.001 }
		check_variable = { party_popularity@market_liberalism > 0.001 }
		check_variable = { party_popularity@military_junta > 0.001 }
		check_variable = { party_popularity@neutrality > 0.001 }
		check_variable = { party_popularity@fascism > 0.001 }
		check_variable = { party_popularity@esotericism > 0.001 }
	}
}

dd_can_select_coalition_partner_communism = {
	dd_has_three_political_parties = yes
	NOT = { has_government = communism }
	OR = {
		has_government = libertarian_socialism
		has_government = syndicate_socialism
		has_government = social_democracy
		has_government = fascism
		has_government = democratic
	}
}

dd_can_select_coalition_partner_libertarian_socialism = {
	dd_has_three_political_parties = yes
	NOT = { has_government = libertarian_socialism }
	OR = {
		has_government = syndicate_socialism

		has_government = social_democracy
	}
}

dd_can_select_coalition_partner_military_junta = {
	dd_has_three_political_parties = yes
	NOT = { has_government = military_junta }
	OR = {
		has_government = market_liberalism
		has_government = neutrality
		has_government = fascism
		AND = {
			tag = HEL
			has_government = democratic
		}
	}
}
