ee_create_left_sr_revolt = {
	create_dynamic_country = {
		original_tag = SOV
		save_global_event_target_as = ee_left_sr_country
		set_country_flag = ee_left_sr_dynamic_rebel
		set_cosmetic_tag = SOV_left_sr_revolt
		set_politics = {
			ruling_party = libertarian_socialism
			elections_allowed = no
		}
		set_popularities = {
			communism = 0
			syndicate_socialism = 5
			libertarian_socialism = 90
			social_democracy = 5
			democratic = 0
			market_liberalism = 0
			military_junta = 0
			fascism = 0
			neutrality = 0
			esotericism = 0
		}
		set_party_name = {
			ideology = libertarian_socialism
			long_name = "Party of Left Socialist-Revolutionaries"
			name = "Left SRs"
		}
		create_country_leader = {
			name = "Maria Spiridonova"
			picture = "Portrait_SOV_Maria_Spiridonova.dds"
			ideology = ee_agrarian_socialism
			expire = "1965.1.1"
			traits = { }
		}
		every_state = {
			limit = {
				OR = {
						state = 239
						state = 249
						state = 250
						state = 251
						state = 252
						state = 256
						state = 401
						state = 582
						state = 651
						state = 652
						state = 829
						state = 833
				}
			}
			add_core_of = event_target:ee_left_sr_country
			event_target:ee_left_sr_country = { transfer_state = PREV }
		}
		SOV = {
			transfer_units_fraction = {
				target = event_target:ee_left_sr_country
				size = 0.40
				stockpile_ratio = 0.40
				army_ratio = 0.40
				navy_ratio = 0.00
				air_ratio = 0.40
			}
		}
		set_capital = { state = 249 }
	}
	event_target:ee_left_sr_country = {
		declare_war_on = { target = SOV type = annex_everything }
	}
	every_country = {
		limit = { has_country_flag = ee_tambov_dynamic_rebel }
		event_target:ee_left_sr_country = {
			declare_war_on = { target = PREV type = annex_everything }
		}
	}
}
