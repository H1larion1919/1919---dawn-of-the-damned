		set_variable = { HEL_race_ionia_support = 54 }

		set_variable = { HEL_race_mainland_support = 58 }

		set_variable = { HEL_race_constantinople_support = 52 }

		10 = {
			set_country_flag = HEL_election_timer_21
			activate_mission = HEL_election_countdown_21
			country_event = { id = hel_election.20 hours = 502 }
		}
		12 = {
			set_country_flag = HEL_election_timer_22
			activate_mission = HEL_election_countdown_22
			country_event = { id = hel_election.20 hours = 526 }
		}
		14 = {
			set_country_flag = HEL_election_timer_23
			activate_mission = HEL_election_countdown_23
			country_event = { id = hel_election.20 hours = 550 }
		}
		16 = {
			set_country_flag = HEL_election_timer_24
			activate_mission = HEL_election_countdown_24
			country_event = { id = hel_election.20 hours = 574 }
		}
		16 = {
			set_country_flag = HEL_election_timer_25
			activate_mission = HEL_election_countdown_25
			country_event = { id = hel_election.20 hours = 598 }
		}
		14 = {
			set_country_flag = HEL_election_timer_26
			activate_mission = HEL_election_countdown_26
			country_event = { id = hel_election.20 hours = 622 }
		}
		10 = {
			set_country_flag = HEL_election_timer_27
			activate_mission = HEL_election_countdown_27
			country_event = { id = hel_election.20 hours = 646 }

HEL_clamp_election_support = {
	if = {
		limit = { check_variable = { HEL_race_mainland_support > 100 } }
		set_variable = { HEL_race_mainland_support = 100 }
	}
	if = {
		limit = { check_variable = { HEL_race_mainland_support < 0 } }
		set_variable = { HEL_race_mainland_support = 0 }
	}
	if = {
		limit = { check_variable = { HEL_race_attica_support > 100 } }
		set_variable = { HEL_race_attica_support = 100 }
	}
	if = {
		limit = { check_variable = { HEL_race_attica_support < 0 } }
		set_variable = { HEL_race_attica_support = 0 }
	}
	if = {
		limit = { check_variable = { HEL_race_south_aegean_support > 100 } }
		set_variable = { HEL_race_south_aegean_support = 100 }
	}
	if = {
		limit = { check_variable = { HEL_race_south_aegean_support < 0 } }
		set_variable = { HEL_race_south_aegean_support = 0 }
	}
	if = {
		limit = { check_variable = { HEL_race_macedonia_support > 100 } }
		set_variable = { HEL_race_macedonia_support = 100 }
	}
	if = {
		limit = { check_variable = { HEL_race_macedonia_support < 0 } }
		set_variable = { HEL_race_macedonia_support = 0 }
	}
	if = {
		limit = { check_variable = { HEL_race_straits_support > 100 } }
		set_variable = { HEL_race_straits_support = 100 }
	}
	if = {
		limit = { check_variable = { HEL_race_straits_support < 0 } }
		set_variable = { HEL_race_straits_support = 0 }
	}
	if = {
		limit = { check_variable = { HEL_race_constantinople_support > 100 } }
		set_variable = { HEL_race_constantinople_support = 100 }
	}
	if = {
		limit = { check_variable = { HEL_race_constantinople_support < 0 } }
		set_variable = { HEL_race_constantinople_support = 0 }
	}
	if = {
		limit = { check_variable = { HEL_race_ionia_support > 100 } }
		set_variable = { HEL_race_ionia_support = 100 }
	}
	if = {
		limit = { check_variable = { HEL_race_ionia_support < 0 } }
		set_variable = { HEL_race_ionia_support = 0 }
	}
	set_variable = { HEL_total_campaign_support = 0 }
	set_variable = { HEL_election_support_temp = HEL_race_mainland_support }
	multiply_variable = { HEL_election_support_temp = 6 }
	add_to_variable = { HEL_total_campaign_support = HEL_election_support_temp }
	set_variable = { HEL_election_support_temp = HEL_race_attica_support }
	multiply_variable = { HEL_election_support_temp = 8 }
	add_to_variable = { HEL_total_campaign_support = HEL_election_support_temp }
	set_variable = { HEL_election_support_temp = HEL_race_south_aegean_support }
	multiply_variable = { HEL_election_support_temp = 7 }
	add_to_variable = { HEL_total_campaign_support = HEL_election_support_temp }
	set_variable = { HEL_election_support_temp = HEL_race_macedonia_support }
	multiply_variable = { HEL_election_support_temp = 9 }
	add_to_variable = { HEL_total_campaign_support = HEL_election_support_temp }
	set_variable = { HEL_election_support_temp = HEL_race_straits_support }
	multiply_variable = { HEL_election_support_temp = 8 }
	add_to_variable = { HEL_total_campaign_support = HEL_election_support_temp }
	set_variable = { HEL_election_support_temp = HEL_race_constantinople_support }
	multiply_variable = { HEL_election_support_temp = 6 }
	add_to_variable = { HEL_total_campaign_support = HEL_election_support_temp }
	set_variable = { HEL_election_support_temp = HEL_race_ionia_support }
	multiply_variable = { HEL_election_support_temp = 6 }
	add_to_variable = { HEL_total_campaign_support = HEL_election_support_temp }
	divide_variable = { HEL_total_campaign_support = 50 }
	round_variable = HEL_total_campaign_support
	set_variable = { HEL_opposition_campaign_support = 100 }
	subtract_from_variable = { HEL_opposition_campaign_support = HEL_total_campaign_support }
	HEL_calculate_election_seats = yes
}
