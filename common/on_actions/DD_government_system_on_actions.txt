on_actions = {
	on_startup = {
		effect = {
			every_country = {
				if = {
					limit = { dd_has_history_government_laws = yes }
					dd_preserve_history_government_laws = yes
				}
				else = {
					dd_assign_ideology_government_laws = yes
				}
				dd_apply_starting_national_spirits = yes
				dd_apply_starting_coalitions = yes
			}
		}
	}

	on_monthly = {
		effect = {
			every_country = {
				dd_refresh_government_system = yes
				dd_refresh_coalition_system = yes
			}
		}
	}

	on_annex = {
		effect = {
			ROOT = {
				set_country_flag = dd_recent_victory
				dd_update_popularity_law = yes
			}
		}
	}

	on_civil_war_end = {
		effect = {
			ROOT = {
				set_country_flag = dd_recent_victory
				dd_update_popularity_law = yes
			}
		}
	}
}
